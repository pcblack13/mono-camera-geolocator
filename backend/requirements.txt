# ── CV (§9.14 "Pinned (CV)") ───────────────────────────────────────────────
# They arrive transitively via the ai_engine path dep; pinned here anyway,
# and matched to the versions verified on the dev box (§0.1) so the container
# and the dev machine cannot disagree about the NumPy ABI.
# `headless` deliberately: the worker image has no X11 and opencv-python would
# drag GUI libs into it. `contrib` deliberately too (2026-08): the CSRT tracker
# in live detection lives in contrib — still ONE cv2 wheel, still no GUI. The
# xfeatures2d ban survives the swap (verify_boundaries.sh §0.1).
numpy==2.4.3
imageio-ffmpeg>=0.6  # static ffmpeg: H.264 preview transcode (app/tasks/transcoding.py)
opencv-contrib-python-headless==4.13.0.92
scipy==1.17.1
# Serial/UART data feeds on EVERY platform (app/services/live_data_service.py):
# pyserial opens COM ports on Windows and /dev/tty* on Linux alike; without it the
# stdlib termios path still serves POSIX and Windows feeds are refused with the
# pip line named. Pure Python, no native build.
pyserial==3.5
# WebSocket data feeds (app/services/live_data_service.py). It arrives anyway as a
# uvicorn[standard] extra, but a feed transport that DEPENDS on it must say so:
# an extra is the server's business and could be dropped without anyone noticing
# the operator's ws:// sender had stopped working.
websockets>=14,<18

# ── geo (§9.14 "Pinned (geo) — FOR THE BACKEND IMAGE ONLY") ────────────────
# Optional on the dev box (system GDAL 3.8.4 substitutes via gis/rasterio_shim.py);
# NOTHING substitutes for them inside the image. Without these, local_orthophoto is
# dead, GeoTIFF uploads are rejected, and total_ce90_m is uncomputable — silently.
# The GDAL *bindings* are NOT pip-installable from a plain wheel: the Dockerfile
# installs libgdal-dev and then `pip install GDAL==$(gdal-config --version)`.
pyproj==3.7.0
rasterio==1.4.3

# ── export writers (gis/exports) ───────────────────────────────────────────
# gis declares these as hard deps (gis/pyproject.toml) but rides the desktop
# as SOURCE with only THIS file installed into the runtime — without them the
# DXF / GeoPackage / Shapefile / PDF exports grey out in the packaged app.
# Pure wheels on linux + win64, py3.12.
ezdxf==1.4.2
reportlab==4.2.5
